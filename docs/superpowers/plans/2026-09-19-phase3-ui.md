# Poker Companion — Phase 3: UI Layer

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire the poker engine and strategy module into a dark-HUD Next.js UI. Output is a running app at `http://localhost:3000` that satisfies all acceptance criteria.

**Architecture:** Single Next.js 15 App Router page. Zustand store holds hand state and triggers recalculation. All poker math is called inline in the page component (no API routes). Tailwind v4 dark theme throughout.

**Tech Stack:** Next.js 15, TypeScript, Tailwind CSS v4, Zustand

**Spec:** `docs/superpowers/specs/2026-09-19-poker-companion-design.md` (sections 6–10, 14)

## Global Constraints

- Currency display: `R` prefix + space + comma-thousands (e.g., `R 3,400`)
- Card suit colors: ♠ ♣ = white text; ♥ ♦ = `text-red-400` (#F87171)
- Decision colors: CALL = `text-green-500`, FOLD = `text-red-500`, CLOSE = `text-amber-400`, RANGE_REQUIRED = `text-gray-400`
- Background: `bg-[#0D0D0F]`, card/panel surfaces: `bg-[#1A1A1F]`, borders: `border-[#2A2A35]`
- Duplicate card selection must be blocked by the store — already-used cards are greyed out and non-interactive in the picker
- `New Hand` resets heroCards, board, pot, amountToCall, isAllIn; retains gameMode, heroStack, villainStack, positions
- Strategy panel displays "Modern baseline" label — never "GTO"
- Tournament mode shows "Chip-EV only" label
- No API routes, no server actions — all calculations are synchronous client-side
- Run dev server with `npm run dev`; verify at `http://localhost:3000`

```ts
import { create } from "zustand";
import { Card, cardKey } from "@/domain/cards";
import { GameMode, Street, Position, PotType } from "@/domain/hand";

export type SlotName =
  | "hero1" | "hero2"
  | "flop1" | "flop2" | "flop3"
  | "turn" | "river";

const SLOT_ORDER: SlotName[] = [
  "hero1", "hero2", "flop1", "flop2", "flop3", "turn", "river",
];

export interface HandStore {
  // Card slots
  slots: Record<SlotName, Card | null>;

  // Money
  pot: number;
  amountToCall: number;
  heroStack: number | undefined;
  villainStack: number | undefined;
  isAllIn: boolean;

  // Game settings
  gameMode: GameMode;

  // Strategy context
  heroPosition: Position | undefined;
  villainPosition: Position | undefined;
  preflopAggressor: "hero" | "villain" | undefined;
  potType: PotType | undefined;
  numPlayers: number;

  // UI state
  activeSlot: SlotName | null;

  // Actions
  selectCard: (card: Card) => void;
  clearSlot: (slot: SlotName) => void;
  setActiveSlot: (slot: SlotName | null) => void;
  setPot: (v: number) => void;
  setAmountToCall: (v: number) => void;
  setHeroStack: (v: number | undefined) => void;
  setVillainStack: (v: number | undefined) => void;
  setIsAllIn: (v: boolean) => void;
  setGameMode: (mode: GameMode) => void;
  setHeroPosition: (p: Position | undefined) => void;
  setVillainPosition: (p: Position | undefined) => void;
  setPreflopAggressor: (a: "hero" | "villain" | undefined) => void;
  setPotType: (t: PotType | undefined) => void;
  setNumPlayers: (n: number) => void;
  newHand: () => void;
}

const EMPTY_SLOTS: Record<SlotName, Card | null> = {
  hero1: null, hero2: null,
  flop1: null, flop2: null, flop3: null,
  turn: null, river: null,
};

export const useHandStore = create<HandStore>((set, get) => ({
  slots: { ...EMPTY_SLOTS },
  pot: 0,
  amountToCall: 0,
  heroStack: undefined,
  villainStack: undefined,
  isAllIn: false,
  gameMode: "cash",
  heroPosition: undefined,
  villainPosition: undefined,
  preflopAggressor: undefined,
  potType: undefined,
  numPlayers: 2,
  activeSlot: "hero1",

  selectCard: (card: Card) => {
    const { activeSlot, slots } = get();
    if (!activeSlot) return;

    // Block duplicate card selection
    const usedKeys = new Set(
      Object.values(slots)
        .filter((c): c is Card => c !== null)
        .map(cardKey)
    );
    if (usedKeys.has(cardKey(card))) return;

    const newSlots = { ...slots, [activeSlot]: card };

    // Auto-advance to next empty slot
    const currentIdx = SLOT_ORDER.indexOf(activeSlot);
    let nextSlot: SlotName | null = null;
    for (let i = currentIdx + 1; i < SLOT_ORDER.length; i++) {
      if (newSlots[SLOT_ORDER[i]] === null) {
        nextSlot = SLOT_ORDER[i];
        break;
      }
    }

    set({ slots: newSlots, activeSlot: nextSlot });
  },

  clearSlot: (slot: SlotName) => {
    const { slots } = get();
    set({ slots: { ...slots, [slot]: null }, activeSlot: slot });
  },

  setActiveSlot: (slot) => set({ activeSlot: slot }),
  setPot: (v) => set({ pot: Math.max(0, v) }),
  setAmountToCall: (v) => set({ amountToCall: Math.max(0, v) }),
  setHeroStack: (v) => set({ heroStack: v }),
  setVillainStack: (v) => set({ villainStack: v }),
  setIsAllIn: (v) => set({ isAllIn: v }),
  setGameMode: (mode) => set({ gameMode: mode }),
  setHeroPosition: (p) => set({ heroPosition: p }),
  setVillainPosition: (p) => set({ villainPosition: p }),
  setPreflopAggressor: (a) => set({ preflopAggressor: a }),
  setPotType: (t) => set({ potType: t }),
  setNumPlayers: (n) => set({ numPlayers: n }),

  newHand: () => {
    const { gameMode, heroStack, villainStack, heroPosition, villainPosition, preflopAggressor, potType, numPlayers } = get();
    set({
      slots: { ...EMPTY_SLOTS },
      pot: 0,
      amountToCall: 0,
      isAllIn: false,
      activeSlot: "hero1",
      // Retain settings
      gameMode,
      heroStack,
      villainStack,
      heroPosition,
      villainPosition,
      preflopAggressor,
      potType,
      numPlayers,
    });
  },
}));

// Selector helpers (pure functions — call outside of Zustand to avoid re-renders)
export function getHeroCards(slots: Record<SlotName, Card | null>): Card[] {
  return [slots.hero1, slots.hero2].filter((c): c is Card => c !== null);
}

export function getBoardCards(slots: Record<SlotName, Card | null>): Card[] {
  return [slots.flop1, slots.flop2, slots.flop3, slots.turn, slots.river]
    .filter((c): c is Card => c !== null);
}

export function getAllUsedCards(slots: Record<SlotName, Card | null>): Card[] {
  return Object.values(slots).filter((c): c is Card => c !== null);
}

export function getCurrentStreet(slots: Record<SlotName, Card | null>): Street {
  const board = getBoardCards(slots);
  if (board.length >= 5) return "river";
  if (board.length >= 4) return "turn";
  if (board.length >= 3) return "flop";
  return "preflop";
}

export function getEffectiveStack(
  heroStack: number | undefined,
  villainStack: number | undefined
): number | undefined {
  if (heroStack !== undefined && villainStack !== undefined) {
    return Math.min(heroStack, villainStack);
  }
  return undefined;
}

export function getSPR(
  effectiveStack: number | undefined,
  pot: number
): number | undefined {
  if (effectiveStack === undefined || pot <= 0) return undefined;
  return effectiveStack / pot;
}
```

```tsx
"use client";

import { SUIT_SYMBOLS, SUIT_TEXT_CLASS_MAP } from "@/domain/cards";
import { useHandStore, getHeroCards, getBoardCards, SlotName } from "@/state/handStore";
import type { Card, Suit } from "@/domain/cards";

// Add SUIT_TEXT_CLASS_MAP to cards.ts if not present — it maps Suit → Tailwind class
// "s" | "c" → "text-white", "h" | "d" → "text-red-400"

function CardSlot({ card, slot, active }: { card: Card | null; slot: SlotName; active: boolean }) {
  const setActiveSlot = useHandStore(s => s.setActiveSlot);
  const clearSlot = useHandStore(s => s.clearSlot);

  const suitClass = card
    ? (card.suit === "h" || card.suit === "d" ? "text-red-400" : "text-white")
    : "text-gray-600";

  function handleClick() {
    if (card) {
      clearSlot(slot);
    } else {
      setActiveSlot(slot);
    }
  }

  return (
    <button
      onClick={handleClick}
      className={[
        "flex flex-col items-center justify-center rounded border w-11 h-14 text-sm font-bold",
        "transition-colors",
        active
          ? "border-blue-500 bg-[#1A1A2F]"
          : "border-[#2A2A35] bg-[#1A1A1F]",
        suitClass,
      ].join(" ")}
    >
      {card ? (
        <>
          <span>{card.rank}</span>
          <span className="text-xs">{SUIT_SYMBOLS[card.suit]}</span>
        </>
      ) : (
        <span className="text-gray-600 text-lg">?</span>
      )}
    </button>
  );
}
```

```tsx
interface MoneyInputProps {
  label: string;

  return `R ${v.toLocaleString("en-ZA")}`;
```

```tsx
"use client";

import type { DecisionResult as DecisionResultType } from "@/poker/decisionEngine";

const SIGNAL_CONFIG = {
  CALL: { emoji: "🟢", label: "CALL", class: "text-green-500" },
  FOLD: { emoji: "🔴", label: "FOLD", class: "text-red-500" },
  CLOSE: { emoji: "🟠", label: "CLOSE / IMPLIED ODDS", class: "text-amber-400" },
  RANGE_REQUIRED: { emoji: "⚪", label: "RANGE REQUIRED", class: "text-gray-400" },
};

function pct(v: number | undefined) {
  if (v === undefined) return "—";
  return `${(v * 100).toFixed(1)}%`;
}

function marginSign(m: number) {
  return m >= 0 ? `+${(m * 100).toFixed(1)}%` : `${(m * 100).toFixed(1)}%`;
}

interface Props {
  result: DecisionResultType;
}

export default function DecisionResult({ result }: Props) {
  const { signal, requiredEquity, comparisonProbability, margin, reason, impliedOdds } = result;
  const cfg = SIGNAL_CONFIG[signal];

  return (
    <div className="space-y-3">
      <div className={`text-3xl font-black ${cfg.class}`}>
        {cfg.emoji} {cfg.label}
      </div>

      <div className="grid grid-cols-2 gap-x-4 gap-y-1 text-sm">
        <span className="text-gray-500">Required equity</span>
        <span className="text-white text-right">{pct(requiredEquity)}</span>

        {comparisonProbability !== undefined && (
          <>
            <span className="text-gray-500">Draw probability</span>
            <span className="text-white text-right">{pct(comparisonProbability)}</span>
          </>
        )}

        {margin !== undefined && (
          <>
            <span className="text-gray-500">Margin</span>
            <span className={`text-right ${margin >= 0 ? "text-green-400" : "text-red-400"}`}>
              {marginSign(margin)}
            </span>
          </>
        )}
      </div>

      {signal === "RANGE_REQUIRED" && (
        <p className="text-xs text-gray-500 italic">{reason}</p>
      )}

      {impliedOdds && signal === "CLOSE" && impliedOdds.futureWinningsRequired < Infinity && (
        <div className="text-xs text-gray-400 border-t border-[#2A2A35] pt-2 mt-2">
          <span>Additional future winnings required: </span>
          <span className="text-white">
            R {Math.round(impliedOdds.futureWinningsRequired).toLocaleString("en-ZA")}
          </span>
          {impliedOdds.effectiveStackAvailable !== undefined && (
            <span className={impliedOdds.possible ? " text-green-400" : " text-red-400"}>
              {" "}({impliedOdds.possible ? "Possible" : "Unlikely"} from stack)
            </span>
          )}
        </div>
      )}
    </div>
  );
}
```

```tsx
        </span>

      </button>

      {open && (
        <div className="space-y-2 text-sm">
          <div className="flex justify-between">
            <span className="text-gray-500">Board</span>
            <span className="text-gray-300">{boardTexture.join(" / ")}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-gray-500">Range advantage</span>
            <span className="text-white capitalize">{rangeAdvantage.toLowerCase()}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-gray-500">Nut advantage</span>
            <span className="text-white capitalize">{nutAdvantage.toLowerCase()}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-gray-500">Strategy</span>
            <span className="text-white">{STYLE_LABELS[style]}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-gray-500">Suggested size</span>
            <span className="text-white">
              {suggestedBetSizes.map(f => `${Math.round(f * 100)}%`).join("–")}
              {pot > 0 && ` (${absoluteSizes.map(v => formatMoney(v)).join("–")})`}
            </span>
          </div>
          <div className="flex justify-between">
            <span className="text-gray-500">Confidence</span>
            <span className={CONFIDENCE_CLASS[confidence] ?? "text-gray-400"}>{confidence}</span>
          </div>
          <p className="text-gray-500 text-xs italic pt-1">{explanation}</p>
        </div>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useMemo } from "react";
import { useHandStore, getHeroCards, getBoardCards, getCurrentStreet, getEffectiveStack, getSPR, SlotName } from "@/state/handStore";
import { calculateOuts } from "@/poker/outs";
import { detectDraws } from "@/poker/draws";
import { evaluateBestHand } from "@/poker/evaluator";
import { calculateDecision } from "@/poker/decisionEngine";
import { analyzeBoardTexture } from "@/strategy/boardTexture";
import { getFlopStrategy, getTurnStrategy, getRiverStrategy } from "@/strategy/flopStrategy";
import type { PreFlopContext } from "@/domain/hand";

import Board from "@/components/Board";
import CardPicker from "@/components/CardPicker";
import MoneyInput from "@/components/MoneyInput";
import DecisionResultPanel from "@/components/DecisionResult";
import DrawSummary from "@/components/DrawSummary";
import StackSummary from "@/components/StackSummary";
import StrategyPanel from "@/components/StrategyPanel";

// Import getTurnStrategy and getRiverStrategy from their own files
import { getTurnStrategy } from "@/strategy/turnStrategy";
import { getRiverStrategy } from "@/strategy/riverStrategy";

export default function Home() {
  const slots = useHandStore(s => s.slots);
  const pot = useHandStore(s => s.pot);
  const amountToCall = useHandStore(s => s.amountToCall);
  const heroStack = useHandStore(s => s.heroStack);
  const villainStack = useHandStore(s => s.villainStack);
  const isAllIn = useHandStore(s => s.isAllIn);
  const gameMode = useHandStore(s => s.gameMode);
  const heroPosition = useHandStore(s => s.heroPosition);
  const villainPosition = useHandStore(s => s.villainPosition);
  const preflopAggressor = useHandStore(s => s.preflopAggressor);
  const potType = useHandStore(s => s.potType);
  const numPlayers = useHandStore(s => s.numPlayers);

  const setPot = useHandStore(s => s.setPot);
  const setAmountToCall = useHandStore(s => s.setAmountToCall);
  const setHeroStack = useHandStore(s => s.setHeroStack);
  const setVillainStack = useHandStore(s => s.setVillainStack);
  const setIsAllIn = useHandStore(s => s.setIsAllIn);
  const setGameMode = useHandStore(s => s.setGameMode);
  const newHand = useHandStore(s => s.newHand);

  const heroCards = getHeroCards(slots);
  const board = getBoardCards(slots);
  const street = getCurrentStreet(slots);
  const effectiveStack = getEffectiveStack(heroStack, villainStack);
  const spr = getSPR(effectiveStack, pot);

  const hand = useMemo(() => ({
    gameMode, heroCards, board, street, pot, amountToCall, heroStack, villainStack, isAllIn,
  }), [gameMode, heroCards, board, street, pot, amountToCall, heroStack, villainStack, isAllIn]);

  const preflopContext: PreFlopContext = useMemo(() => ({
    heroPosition, villainPosition, preflopAggressor, potType, numPlayers,
  }), [heroPosition, villainPosition, preflopAggressor, potType, numPlayers]);

  // All poker calculations — synchronous, < 100 ms
  const { outsResult, drawAnalysis, currentHand, decisionResult, boardAnalysis, strategyResult } = useMemo(() => {
    const outsResult = calculateOuts(heroCards, board);
    const drawAnalysis = detectDraws(heroCards, board);
    const currentHand = heroCards.length > 0 ? evaluateBestHand(heroCards, board) : null;

    const decisionResult = board.length >= 3 && amountToCall >= 0
      ? calculateDecision(hand, drawAnalysis, outsResult)
      : null;

    const boardAnalysis = board.length >= 3 ? analyzeBoardTexture(board) : null;

    const strategyResult = boardAnalysis
      ? street === "river"
        ? getRiverStrategy(preflopContext, boardAnalysis, spr ?? 0)
        : street === "turn"
        ? getTurnStrategy(preflopContext, boardAnalysis, spr ?? 0)
        : getFlopStrategy(preflopContext, boardAnalysis, spr ?? 0)
      : null;

    return { outsResult, drawAnalysis, currentHand, decisionResult, boardAnalysis, strategyResult };
  }, [heroCards, board, hand, preflopContext, amountToCall, street, spr]);

  return (
    <main className="max-w-md mx-auto px-4 py-6 space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div className="flex rounded overflow-hidden border border-[#2A2A35]">
          {(["cash", "tournament"] as const).map(mode => (
            <button
              key={mode}
              onClick={() => setGameMode(mode)}
              className={[
                "px-3 py-1 text-xs uppercase tracking-widest font-bold transition-colors",
                gameMode === mode
                  ? "bg-[#2A2A3F] text-white"
                  : "bg-[#1A1A1F] text-gray-500 hover:text-gray-300",
              ].join(" ")}
            >
              {mode}
              {mode === "tournament" && gameMode === "tournament" && (
                <span className="ml-1 text-gray-600 normal-case tracking-normal font-normal">(Chip EV)</span>
              )}
            </button>
          ))}
        </div>
        <button
          onClick={newHand}
          className="px-4 py-1.5 text-xs uppercase tracking-widest font-bold border border-[#2A2A35] rounded bg-[#1A1A1F] text-gray-300 hover:bg-[#2A2A3F] transition-colors"
        >
          New Hand
        </button>
      </div>

      {/* Cards */}
      <Board />

      {/* Card Picker */}
      <CardPicker />

      {/* Money Inputs */}
      <div className="space-y-3">
        <MoneyInput label="Pot" value={pot} onChange={setPot} />
        <MoneyInput label="Call" value={amountToCall} onChange={setAmountToCall} />

        <div className="flex items-center gap-3">
          <span className="text-xs text-gray-400 uppercase tracking-widest w-16 shrink-0">All-In</span>
          <button
            onClick={() => setIsAllIn(!isAllIn)}
            className={[
              "w-10 h-6 rounded-full transition-colors border",
              isAllIn ? "bg-blue-600 border-blue-500" : "bg-[#1A1A1F] border-[#2A2A35]",
            ].join(" ")}
          >
            <span
              className={[
                "block w-4 h-4 mx-1 rounded-full bg-white transition-transform",
                isAllIn ? "translate-x-4" : "translate-x-0",
              ].join(" ")}
            />
          </button>
        </div>
      </div>

      {/* Stack Inputs (optional) */}
      <details className="group">
        <summary className="text-xs text-gray-600 uppercase tracking-widest cursor-pointer hover:text-gray-400">
          Stack sizes (optional)
        </summary>
        <div className="mt-3 space-y-3">
          <MoneyInput label="My Stack" value={heroStack ?? 0} onChange={v => setHeroStack(v || undefined)} />
          <MoneyInput label="Villain" value={villainStack ?? 0} onChange={v => setVillainStack(v || undefined)} />
        </div>
      </details>

      {/* Divider */}
      {board.length >= 3 && <hr className="border-[#2A2A35]" />}

      {/* Decision Panel */}
      {board.length >= 3 && decisionResult && (
        <div className="space-y-4">
          <p className="text-xs text-gray-500 uppercase tracking-widest">Decision</p>
          <DecisionResultPanel result={decisionResult} />

          {currentHand && (
            <p className="text-xs text-gray-500">
              Current hand: <span className="text-gray-300">{currentHand.description}</span>
            </p>
          )}

          <DrawSummary drawAnalysis={drawAnalysis} outsResult={outsResult} />
          <StackSummary />
        </div>
      )}

      {/* Strategy Panel */}
      {strategyResult && (
        <StrategyPanel strategy={strategyResult} pot={pot} />
      )}
    </main>
  );
}
```
